function renderTraining(key){

function openTrainingMode(mode){

// جست‌وجوی مقاوم برای WebView اندروید؛ یکسان‌سازی حروف فارسی/عربی و اعداد
function normalizeSurveySearchText(value){
 return String(value==null?'':value)
  .replace(/[يى]/g,'ی').replace(/ك/g,'ک').replace(/[ۀة]/g,'ه')
  .replace(/[أإٱ]/g,'ا').replace(/[۰-۹]/g,function(c){return '۰۱۲۳۴۵۶۷۸۹'.indexOf(c);})
  .replace(/[٠-٩]/g,function(c){return '٠١٢٣٤٥٦٧٨٩'.indexOf(c);})
  .toLowerCase().replace(/\s+/g,' ').trim();
}
function filterSurveyLessons(query){
 var root=document.getElementById('surveyLessonResults');
 var status=document.getElementById('surveyLessonSearchStatus');
 if(!root){console.warn('Survey search: lesson results container not found');return false;}
 if(!status){status=document.createElement('div');status.id='surveyLessonSearchStatus';status.setAttribute('aria-live','polite');root.parentNode.insertBefore(status,root);}
 var q=normalizeSurveySearchText(query);
 var lessons=root.querySelectorAll('.training-lesson');var shown=0;
 for(var i=0;i<lessons.length;i++){
  var text=normalizeSurveySearchText(lessons[i].innerText||lessons[i].textContent);
  var match=!q || text.indexOf(q)!==-1;
  lessons[i].style.display=match?'':'none';
  if(match)shown++;
 }
 status.textContent=q?(shown+' درس از '+lessons.length+' درس پیدا شد.'):('همه '+lessons.length+' درس نمایش داده می‌شوند.');
 return false;
}

function directSaveBlob(name,blob){
 name=String(name||'MILADGEO_EXPORT.bin').replace(/[\\/:*?"<>|]+/g,'_');
 try{
  if(window.AndroidBridge&&typeof window.AndroidBridge.saveFile==='function'&&blob){
   showToast('⏳ آماده‌سازی فایل...');
   var readPromise=(typeof blob.arrayBuffer==='function')
     ? blob.arrayBuffer()
     : new Promise(function(resolve,reject){try{var fr=new FileReader();fr.onload=function(){resolve(fr.result);};fr.onerror=reject;fr.readAsArrayBuffer(blob);}catch(e){reject(e);}});
   readPromise.then(function(buf){
     var bytes=new Uint8Array(buf), binary='', chunk=0x8000;
     for(var i=0;i<bytes.length;i+=chunk){binary+=String.fromCharCode.apply(null,bytes.subarray(i,Math.min(i+chunk,bytes.length)));}
     var base64=btoa(binary);
     return window.AndroidBridge.saveFile(name, blob.type||'application/octet-stream', base64);
   }).then(function(ok){
     if(ok!==false)showToast('✓ File saved: '+name);
     else showToast('Export cancelled');
   }).catch(function(e){
     console.error('Android save failed',e);
     browserSaveBlob(name,blob,false);
   });
   return true;
  }
  if(window.miladgeoNative&&typeof window.miladgeoNative.saveFile==='function'&&blob){
   showToast('⏳ آماده‌سازی فایل...');
   var readPromise2=(typeof blob.arrayBuffer==='function')
     ? blob.arrayBuffer()
     : new Promise(function(resolve,reject){try{var fr2=new FileReader();fr2.onload=function(){resolve(fr2.result);};fr2.onerror=reject;fr2.readAsArrayBuffer(blob);}catch(e){reject(e);}});
   readPromise2.then(function(buf){
     var bytes=Array.from(new Uint8Array(buf));
     return window.miladgeoNative.saveFile({name:name,bytes:bytes});
   }).then(function(ok){
     if(ok)showToast('✓ File saved: '+name);
     else showToast('Export cancelled');
   }).catch(function(e){
     console.error('Native save failed',e);
     browserSaveBlob(name,blob,false);
   });
   return true;
  }
  var url=URL.createObjectURL(blob),a=document.createElement('a');
  a.href=url;a.download=name;a.rel='noopener';a.style.display='none';
  document.body.appendChild(a);a.click();
  setTimeout(function(){try{a.remove();}catch(e){}try{URL.revokeObjectURL(url);}catch(e){}},15000);
  showToast('✓ '+name+' آماده شد');
  return true;
 }catch(e){console.error('Export failed',e);showToast('Export failed');return false;}
}

function gsi16DataWord(wi,info,sign,value){

 return '*'+String(wi).padStart(2,'0')+String(info==null?'00':info).padStart(4,'0')+String(sign||'+').slice(0,1)+raw;
}
function gsi16Scaled(v){
 var n=Number(v);
 if(!isFinite(n))return null;
 var scaled=Math.round(n*1000);
 var sign=scaled<0?'-':'+';
 var abs=Math.abs(scaled).toString();
 if(abs.length>16)return null;
 return {sign:sign,data:abs.padStart(16,'0')};
}

}

function mg180GsiCoordWord(wi,v,width){

 return String(wi).padStart(2,'0')+'....'+(n<0?'-':'+')+data.padStart(width,'0');
}

function crc32(bytes){
 var table=crc32.table,crc=0xFFFFFFFF,i,j;
 if(!table){table=[];for(i=0;i<256;i++){var c=i;for(j=0;j<8;j++)c=(c&1)?(0xEDB88320^(c>>>1)):(c>>>1);table[i]=c>>>0;}crc32.table=table;}
 for(i=0;i<bytes.length;i++)crc=(table[(crc^bytes[i])&255]^(crc>>>8))>>>0;
 return (crc^0xFFFFFFFF)>>>0;
}

function v5DetectDelimiter(line){
 if(line.indexOf('\t')>=0)return '\t';
 if(line.indexOf(',')>=0)return ',';
 if(line.indexOf(';')>=0)return ';';
 return 'space';
}
function v5Split(line,delim){
 line=String(line||'').trim();
 if(!line)return [];
 if(delim==='space')return line.split(/\s+/);
 return line.split(delim).map(function(x){return x.trim();});
}
function v5Num(v){var s=String(v==null?'':v).trim().replace(/^\(|\)$/g,'');if(s==='')return NaN;return Number(s.replace(/,/g,''));}
function v5LooksHeader(a){if(!a||a.length<3)return false;var z=a.slice(0,5).join(' ').toLowerCase();var headerWord=/\b(point|pointno|pointnumber|easting|northing|east|north|elev|elevation|height|code|desc|description|x|y|z)\b/.test(z);var numericCore=a.length>=4&&!isNaN(v5Num(a[1]))&&!isNaN(v5Num(a[2]))&&!isNaN(v5Num(a[3]));return headerWord&&!numericCore;}
function v5ParseText(text,forced){
 var lines=String(text||'').split(/\r?\n/).map(function(x){return x.trim();}).filter(function(x){return x!=='' && x.charAt(0)!=='#' && x.charAt(0)!==';'});
 if(!lines.length)return {rows:[],bad:0,delimiter:'-',format:'-'};
 var first=lines[0],delim=v5DetectDelimiter(first),format=forced&&forced!=='auto'?forced:'auto';
 var start=0,rows=[],bad=0;
 if(v5LooksHeader(v5Split(first,delim)))start=1;
 if(format==='auto'){
   var a=v5Split(lines[start]||'',delim), nums=a.map(v5Num);
   if(a.length>=5 && !isNaN(nums[1]) && !isNaN(nums[2]) && !isNaN(nums[3])){
     format='penzd';
     if(/north/i.test(a[1])||/northing/i.test(a[1]))format='pnezd';
   }else if(a.length>=4 && !isNaN(nums[1]) && !isNaN(nums[2]) && !isNaN(nums[3]))format='penzd';
   else if(a.length>=3 && !isNaN(nums[0]) && !isNaN(nums[1]) && !isNaN(nums[2]))format='xyz';
 }
 for(var i=start;i<lines.length;i++){
   var parts=v5Split(lines[i],delim);if(!parts.length)continue;
   var pno='',dp='',x=NaN,y=NaN,z=0;
   if(format==='penzd' || format==='pnezd'){
     if(parts.length<4){bad++;continue;}
     pno=parts[0];var a=v5Num(parts[1]),b=v5Num(parts[2]);z=v5Num(parts[3]);
     if(format==='penzd'){x=a;y=b;}else{y=a;x=b;}
     dp=parts.slice(4).join(delim==='space'?' ':' ');
   }else if(format==='nez' || format==='enz'){
     if(parts.length<3){bad++;continue;}var a2=v5Num(parts[0]),b2=v5Num(parts[1]);z=v5Num(parts[2]);
     if(format==='nez'){y=a2;x=b2;}else{x=a2;y=b2;}
     pno='P'+String(rows.length+1).padStart(3,'0');dp=parts.slice(3).join(' ');
   }else if(format==='xyz'){
     if(parts.length<3){bad++;continue;}x=v5Num(parts[0]);y=v5Num(parts[1]);z=v5Num(parts[2]);pno=parts[3]||('P'+String(rows.length+1).padStart(3,'0'));dp=parts.slice(4).join(' ');
   }else{bad++;continue;}
   if(!isFinite(x)||!isFinite(y)||!isFinite(z)){bad++;continue;}
   rows.push({no:pno||('P'+String(rows.length+1).padStart(3,'0')),dp:dp||'',x:x,y:y,z:z});
 }
 return {rows:rows,bad:bad,delimiter:delim,format:format,lines:lines.length};
}

function v5ImportPoints(){
 if(!v5ImportText){showToast('Choose a TXT/CSV file first');return false;}
 var f=$("v5ImportFormat")?$("v5ImportFormat").value:'auto',p=v5ParseText(v5ImportText,f);
 if(!p.rows.length){showToast('No valid survey points found');return false;}
 var importKey=v10ImportRegistryKey();
 if(v5ImportedFileKeys[importKey]){showToast('این فایل قبلاً برای این JOB وارد شده است ✓');return false;}
 var autoNo=$("v5PointNoMode")&&$("v5PointNoMode").value==='auto',base=getNextPointNumber(),existingNo={},existingCoord={};
 for(var e=0;e<points.length;e++){var ep=points[e];existingNo[String(ep.no||'').trim().toUpperCase()]=true;existingCoord=[Number(ep.x).toFixed(6),Number(ep.y).toFixed(6),Number(ep.z).toFixed(6)].join('|');existingCoord=existingCoord;}
 /* Build coordinate keys separately; DP is intentionally ignored so the same survey point cannot be duplicated. */
 var coordSet={};
 for(var e2=0;e2<points.length;e2++){var ep2=points[e2];coordSet[[Number(ep2.x).toFixed(6),Number(ep2.y).toFixed(6),Number(ep2.z).toFixed(6)].join('|')]=true;}
 var added=0,dupes=0;
 for(var i=0;i<p.rows.length;i++){
   var r=p.rows[i],no=autoNo?'P'+String(base+added).padStart(3,'0'):String(r.no||('P'+String(base+added).padStart(3,'0')));
   var coordKey=[Number(r.x).toFixed(6),Number(r.y).toFixed(6),Number(r.z).toFixed(6)].join('|');
   var noKey=no.trim().toUpperCase();
   if(coordSet[coordKey] || existingNo[noKey]){dupes++;continue;}
   points.push({id:Date.now()+Math.floor(Math.random()*100000)+i,no:no,dp:r.dp||'',x:r.x,y:r.y,z:r.z,distance:null,azimuth:null,textImport:true});
   coordSet[coordKey]=true;existingNo[noKey]=true;added++;
 }
 v5ImportedFileKeys[importKey]=true;v10SaveImportRegistry();
 window.points=points;saveJSON(POINTS_KEY,points);refreshPoints();updateAllUI();try{window.m4pRender&&window.m4pRender();window.m4pRenderControl&&window.m4pRenderControl();window.m4pRenderStats&&window.m4pRenderStats();}catch(e){}
 showToast(added+' point(s) imported ✓'+(dupes?' — '+dupes+' duplicate point(s) skipped':'')+(p.bad?' — '+p.bad+' bad row(s) skipped':'')+(added===0?' — file marked as imported':'') );
 return false;
}
