function esc(v){return String(v==null?'':v).replace(/[&<>"']/g,function(c){return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]})}
function num(v){var x=Number(String(v==null?'':v).replace(/,/g,'').trim());return isFinite(x)?x:NaN}

function normalizeName(v,i){v=String(v==null?'':v).trim();return v||('P'+(i+1))}
function stake3SearchNormalize(v){
  return String(v==null?'':v).trim().toLowerCase().replace(/\s+/g,'').replace(/^p(?=\d)/,'');
}
function stake3SearchPoint(){
  var inp=q('stake3SearchInput'),box=q('stake3SearchResult');if(!inp||!box)return false;
  var query=stake3SearchNormalize(inp.value);
  if(!query){box.className='stake3-search-result show';box.innerHTML='<div class="stake3-search-notfound">شماره یا نام نقطه را وارد کن.</div>';return false;}
  var found=-1;
  for(var si=0;si<list.length;si++){
    var nm=stake3SearchNormalize(normalizeName(list[si].name,si));
    if(nm===query || nm.indexOf(query)>=0){found=si;break;}
  }
  if(found<0){
    box.className='stake3-search-result show';
    box.innerHTML='<div class="stake3-search-notfound">نقطه «'+esc(inp.value)+'» پیدا نشد.</div>';
    return false;
  }
  activeIndex=found;
  render();
  var p=list[found],name=normalizeName(p.name,found);
  var coordHtml='';
  if(p.kind==='geo'){
    coordHtml='<div class="stake3-selected-cell"><small>طول</small><b>'+Number(p.lng).toFixed(6)+'</b></div>'+
              '<div class="stake3-selected-cell"><small>عرض</small><b>'+Number(p.lat).toFixed(6)+'</b></div>';
  }else{
    coordHtml='<div class="stake3-selected-cell"><small>X / Easting</small><b>'+Number(p.x).toFixed(2)+'</b></div>'+
              '<div class="stake3-selected-cell"><small>Y / Northing</small><b>'+Number(p.y).toFixed(2)+'</b></div>';
  }
  box.className='stake3-search-result show';
  box.innerHTML='<div class="stake3-selected-head"><span class="stake3-selected-name">'+esc(name)+'</span><span class="stake3-selected-badge">'+(p.done?'✓ پیاده شده':'نقطه انتخاب شد')+'</span></div>'+
    '<div class="stake3-selected-grid">'+coordHtml+
    '<div class="stake3-selected-cell"><small>شماره در لیست</small><b>'+fa(found+1)+'</b></div></div>'+
    '<div class="stake3-selected-action"><button class="btn primary" type="button" id="stake3SearchGo">'+(p.done?'✓ این نقطه قبلاً پیاده شده':'🎯 پیاده‌سازی '+esc(name))+'</button></div>';
  var go=q('stake3SearchGo');
  if(go&&!p.done)go.addEventListener('click',function(e){e.preventDefault();startIndex(found);});
  setTimeout(function(){
    var row=box.closest ? null : null;
    var listBox=q('stake3List');
    if(listBox){
      var target=listBox.querySelector('[data-stake3-index="'+found+'"]');
      if(target&&target.parentElement)target.parentElement.scrollIntoView({behavior:'smooth',block:'center'});
    }
  },40);
  return false;
}
function render(){
  var box=q('stake3List'),prog=q('stake3Progress');if(!box)return;
  var done=list.filter(function(p){return p.done}).length;
  if(prog)prog.textContent=fa(done)+' / '+fa(list.length)+' نقطه پیاده شده';
  if(!list.length){box.innerHTML='<div class="stake3-empty">هنوز نقطه‌ای برای پیاده‌سازی وارد نشده است.</div>';return;}
  box.innerHTML=list.map(function(p,i){
    var status=p.done?'✓':(i===activeIndex?'🎯':'⏳');
    var cls='stake3-row'+(p.done?' done':'')+(i===activeIndex?' active':'');
    var coord=p.kind==='geo'?(Number(p.lat).toFixed(6)+' , '+Number(p.lng).toFixed(6)):(Number(p.x).toFixed(2)+' , '+Number(p.y).toFixed(2));
    var action=p.done?'<span class="stake3-status">✓</span>':'<button class="btn primary stake3-go" type="button" data-stake3-index="'+i+'">پیاده‌سازی</button>';
    return '<div class="'+cls+'"><div class="stake3-status">'+status+'</div><div class="stake3-point">'+esc(normalizeName(p.name,i))+'</div><div class="stake3-coord">'+esc(coord)+'</div><div>'+action+'</div></div>';
  }).join('');
  box.querySelectorAll('[data-stake3-index]').forEach(function(b){b.addEventListener('click',function(e){e.preventDefault();e.stopPropagation();startIndex(Number(b.getAttribute('data-stake3-index')));});});
}
function fa(v){return String(v).replace(/\d/g,function(x){return '۰۱۲۳۴۵۶۷۸۹'[x]})}
function startIndex(i){
  if(!list[i]||list[i].done)return false;
  var p=list[i];activeIndex=i;
  if(p.kind==='utm'){
    var zone=num(p.zone);if(!isFinite(zone)||zone<1||zone>60){showToast('UTM Zone را برای نقاط PenZD صحیح وارد کن');return false;}
    if(typeof window.m17ConvertUtmToWgs84!=='function'){showToast('ارتباط با نقشه منوی ۱۷ آماده نیست');return false;}
    /* V309: PenZD UTM targets must use their own north/south hemisphere.
   Do not derive the hemisphere from the current GPS/map position. */
    var northern=Number(p.y)<10000000;
    var c=(typeof window.m17UtmToWgs84==='function')
      ? window.m17UtmToWgs84(Number(p.x),Number(p.y),zone,northern)
      : window.m17ConvertUtmToWgs84(Number(p.x),Number(p.y),zone);
    if(!c||!isFinite(c.lat)||!isFinite(c.lng)){showToast('تبدیل مختصات نقطه انجام نشد');return false;}
    p.lat=c.lat;p.lng=c.lng;
  }
  if(!isFinite(Number(p.lat))||!isFinite(Number(p.lng))){showToast('مختصات نقطه معتبر نیست');return false;}
  window.__stake3ActiveIndex=i;
  showPage('menu17');
  setTimeout(function(){try{if(typeof m17Init==='function')m17Init();if(typeof window.m17LiveOpen!=='function'){showToast('پیاده‌سازی زنده آماده نیست');return;}window.m17LiveOpen({lat:Number(p.lat),lng:Number(p.lng)},normalizeName(p.name,i));render();}catch(e){console.error('STAKE3_START',e);showToast('شروع پیاده‌سازی انجام نشد');}},180);
  return false;
}
window.stake3LiveFinished=function(target){
  var i=Number(window.__stake3ActiveIndex);if(!isFinite(i)||!list[i])return;
  var p=list[i];if(target){p.lat=Number(target.lat);p.lng=Number(target.lng)}p.done=true;
  try{
    var saved=window.miladGeoAddPointToJob({name:normalizeName(p.name,i),dp:p.dp||'STAKE OUT',x:p.x,y:p.y,z:p.z||0,lat:p.lat,lng:p.lng});
    if(!saved)console.warn('Stake-out point already exists in JOB or could not be converted');
  }catch(e){console.error('STAKE3_FINISH_JOB',e);}
  activeIndex=-1;window.__stake3ActiveIndex=-1;render();
  if(typeof showToast==='function')showToast(normalizeName(p.name,i)+' ✓ پیاده شد');
  setTimeout(function(){try{showPage('realStake');}catch(e){}},80);
  var next=-1;for(var j=i+1;j<list.length;j++)if(!list[j].done){next=j;break;}if(next<0)for(var k=0;k<list.length;k++)if(!list[k].done){next=k;break;}
  if(next>=0){activeIndex=next;render();}
};
