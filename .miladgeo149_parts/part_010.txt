function excelRows(rows){
  var out=[];if(!rows||!rows.length)return out;
  function norm(v){return String(v==null?'':v).trim().toLowerCase().replace(/[\s_\-\/\\().:]/g,'')}
  function idxBy(names,header){
    /* V348: resolve exact column aliases first.  Single-letter aliases
       such as N/E/X/Y must never match a longer word by substring
       (e.g. "Easting" contains "n", which previously mapped Northing
       to the Easting column in real PENZD Excel files). */
    for(var i=0;i<header.length;i++){
      var h=norm(header[i]);
      for(var j=0;j<names.length;j++){
        if(h===norm(names[j]))return i;
      }
    }
    /* Only use substring matching for descriptive aliases with 3+ chars. */
    for(var i2=0;i2<header.length;i2++){
      var h2=norm(header[i2]);
      for(var j2=0;j2<names.length;j2++){
        var n2=norm(names[j2]);
        if(n2.length>=3&&(h2.indexOf(n2)>=0||n2.indexOf(h2)>=0))return i2;
      }
    }
    return -1
  }
  var first=rows[0]||[], header=first.map(norm);
  var hasHeader=false;
  for(var h=0;h<header.length;h++){if(/^(point|pt|p|easting|east|x|northing|north|n|y|elevation|elev|z|description|desc|dp|code|d)$/.test(header[h])){hasHeader=true;break;}}
  var start=hasHeader?1:0;
  /* V349: for a simple 2-column coordinate Excel/CSV, the contract is explicit:
     column 1 = X/Easting, column 2 = Y/Northing. No PENZD guessing is allowed.
     Standard headered PENZD and headerless PNE/PENZD files keep their existing mapping. */
  var simpleXY = !hasHeader && first.length===2;
  var ip=hasHeader?idxBy(['point','pt','name','id','code','p'],header):-1;
  var ix=hasHeader?idxBy(['easting','east','x','e'],header):(simpleXY?0:1);
  var iy=hasHeader?idxBy(['northing','north','y','n'],header):(simpleXY?1:2);
  var iz=hasHeader?idxBy(['elevation','elev','height','z'],header):(simpleXY?-1:3);
  var id=hasHeader?idxBy(['description','desc','descriptionpoint','dp','code','d'],header):(simpleXY?-1:4);
  /* If the header is non-standard or mapping is incomplete, use the standard PENZD order P/E/N/Z/D. */
  if(ip<0 && !simpleXY)ip=0;
  if(ix<0)ix=simpleXY?0:1;
  if(iy<0)iy=simpleXY?1:2;
  if(iz<0 && !simpleXY)iz=3;
  for(var r=start;r<rows.length;r++){
    var row=rows[r]||[];
    if(!row.length)continue;
    var x=num(row[ix]),y=num(row[iy]);
    /* V347: many real survey exports have no header and are written as
       Point / Northing / Easting / Z / Description (PNEZD).
       The old fallback assumed P/E/N, which reversed the coordinates.
       Detect the characteristic UTM ranges only for headerless rows:
       Northing is normally > 1,000,000 while Easting is typically
       100,000–900,000. Headered files keep their explicit column mapping. */
    if(!hasHeader && isFinite(x) && isFinite(y) &&
       Math.abs(x)>1000000 && Math.abs(y)>=100000 && Math.abs(y)<=1000000){
      var tmpNoHeader=x;x=y;y=tmpNoHeader;
    }
    /* Headered survey files can explicitly arrive as Northing / Easting. */
    if(hasHeader&&header[ix]&&/(^n|north)/.test(header[ix])&&header[iy]&&/(^e|east)/.test(header[iy])){
      var tmp=x;x=y;y=tmp;
    }
    var z=iz<row.length?num(row[iz]):NaN;
    if(!isFinite(x)||!isFinite(y))continue;
    out.push({name:normalizeName(row[ip],out.length),kind:'utm',x:x,y:y,z:isFinite(z)?z:0,dp:id>=0&&id<row.length?String(row[id]==null?'':row[id]).trim():'',zone:num(q('stake3Zone')?q('stake3Zone').value:'39'),done:false});
  }
  return out;
}

  };
  r.onerror=function(){setStatus('خطا در خواندن فایل Excel');showToast('خواندن Excel ناموفق بود')};
  r.readAsArrayBuffer(file);

function m4ProfileDistance(a,b){
  var R=6378137,lat1=Number(a.lat)*Math.PI/180,lat2=Number(b.lat)*Math.PI/180,dlat=lat2-lat1,dlon=(Number(b.lng)-Number(a.lng))*Math.PI/180;
  var h=Math.sin(dlat/2)*Math.sin(dlat/2)+Math.cos(lat1)*Math.cos(lat2)*Math.sin(dlon/2)*Math.sin(dlon/2);
  return 2*R*Math.atan2(Math.sqrt(h),Math.sqrt(Math.max(0,1-h)));
}
function m4ProfileInterpolate(a,b,t){return {lat:Number(a.lat)+(Number(b.lat)-Number(a.lat))*t,lng:Number(a.lng)+(Number(b.lng)-Number(a.lng))*t};}
function m4ProfileSample(lat,lng){
  if(!m4Last||!m4Last.raster)return NaN;
  var bb=m4Last.rasterBbox,w=m4Last.width,h=m4Last.height,g=m4Last.raster;
  if(!bb||w<2||h<2)return NaN;
  var fx=(Number(lng)-bb[0])/(bb[2]-bb[0])*(w-1),fy=(bb[3]-Number(lat))/(bb[3]-bb[1])*(h-1);
  if(fx<0||fy<0||fx>w-1||fy>h-1)return NaN;
  var x0=Math.floor(fx),y0=Math.floor(fy),x1=Math.min(w-1,x0+1),y1=Math.min(h-1,y0+1),tx=fx-x0,ty=fy-y0;
  var z00=Number(g[y0*w+x0]),z10=Number(g[y0*w+x1]),z01=Number(g[y1*w+x0]),z11=Number(g[y1*w+x1]);
  if(!isFinite(z00)||!isFinite(z10)||!isFinite(z01)||!isFinite(z11))return NaN;
  return z00*(1-tx)*(1-ty)+z10*tx*(1-ty)+z01*(1-tx)*ty+z11*tx*ty;
}
