 function mgAgentDigits(raw){return String(raw||'').replace(/[۰-۹]/g,function(c){return String('۰۱۲۳۴۵۶۷۸۹'.indexOf(c));}).replace(/[٠-٩]/g,function(c){return String('٠١٢٣٤٥٦٧٨٩'.indexOf(c));}).replace(/[−–]/g,'-').replace(/[٫]/g,'.');}
 function mgAgentParseCoordPairs(raw,zone){
  var s=mgAgentDigits(raw),pairs=[],m;
  var re=/\(\s*(-?\d+(?:\.\d+)?)\s*[,،;|]\s*(-?\d+(?:\.\d+)?)\s*\)/g;
  while((m=re.exec(s))!==null)pairs.push({x:Number(m[1]),y:Number(m[2])});
  if(pairs.length)return pairs;
  var x=s.match(/(?:\bX\b|easting|شرقی)\s*[:=]?\s*(-?\d+(?:\.\d+)?)/i);
  var y=s.match(/(?:\bY\b|northing|شمالی)\s*[:=]?\s*(-?\d+(?:\.\d+)?)/i);
  var lon=s.match(/(?:\blon(?:gitude)?\b|\blng\b|طول(?:\s*جغرافیایی)?)\s*[:=]?\s*(-?\d+(?:\.\d+)?)/i);
  var lat=s.match(/(?:\blat(?:itude)?\b|عرض(?:\s*جغرافیایی)?)\s*[:=]?\s*(-?\d+(?:\.\d+)?)/i);
  if(x&&y)return [{x:Number(x[1]),y:Number(y[1])}];
  if(lon&&lat)return [{x:Number(lon[1]),y:Number(lat[1])}];
  var cleaned=s.replace(/(?:زون|zone)\s*[:=]?\s*\d{1,2}/ig,' ');
  var nums=(cleaned.match(/-?\d+(?:\.\d+)?/g)||[]).map(Number);
  if(nums.length===2)return [{x:nums[0],y:nums[1]}];
  return [];
 }
