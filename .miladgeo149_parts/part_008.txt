  function m17UtmToWgs84(x,y,zone,northern){
    var a=6378137.0,eccSq=0.0066943799901413165,k0=0.9996,eccPrimeSq=eccSq/(1-eccSq);
    var e1=(1-Math.sqrt(1-eccSq))/(1+Math.sqrt(1-eccSq)),x0=Number(x)-500000,y0=Number(y);
    if(!northern)y0-=10000000;
    var M=y0/k0,mu=M/(a*(1-eccSq/4-3*eccSq*eccSq/64-5*eccSq*eccSq*eccSq/256));
    var J1=3*e1/2-27*Math.pow(e1,3)/32,J2=21*e1*e1/16-55*Math.pow(e1,4)/32,J3=151*Math.pow(e1,3)/96,J4=1097*Math.pow(e1,4)/512;
    var fp=mu+J1*Math.sin(2*mu)+J2*Math.sin(4*mu)+J3*Math.sin(6*mu)+J4*Math.sin(8*mu);
    var C1=eccPrimeSq*Math.cos(fp)*Math.cos(fp),T1=Math.tan(fp)*Math.tan(fp),N1=a/Math.sqrt(1-eccSq*Math.sin(fp)*Math.sin(fp));
    var R1=a*(1-eccSq)/Math.pow(1-eccSq*Math.sin(fp)*Math.sin(fp),1.5),D=x0/(N1*k0);
    var lat=fp-(N1*Math.tan(fp)/R1)*(D*D/2-(5+3*T1+10*C1-4*C1*C1-9*eccPrimeSq)*Math.pow(D,4)/24+(61+90*T1+298*C1+45*T1*T1-252*eccPrimeSq-3*C1*C1)*Math.pow(D,6)/720);
    var lon0=((Number(zone)-1)*6-180+3)*Math.PI/180;
    var lon=lon0+(D-(1+2*T1+C1)*Math.pow(D,3)/6+(5-2*C1+28*T1-3*C1*C1+8*eccPrimeSq+24*T1*T1)*Math.pow(D,5)/120)/Math.cos(fp);
    return {lat:lat*180/Math.PI,lng:lon*180/Math.PI};
  }

  function requestBrowserGPS(){
    if(!navigator.geolocation){showGpsPrompt();return;}
    navigator.geolocation.getCurrentPosition(function(pos){
      useLocation(pos.coords.latitude,pos.coords.longitude);
      hideGpsPrompt();
    },function(){
      showGpsPrompt();
    }, {enableHighAccuracy:true,timeout:12000,maximumAge:0});
  }

  function requestGPS(){
    /* V191: Android app first handles permission, then the user can manually turn GPS on. */
    try{
      if(window.AndroidBridge && typeof window.AndroidBridge.requestLocationPermission==='function'){
        window.AndroidBridge.requestLocationPermission();
        return;
      }
    }catch(_){ }
    if(navigator.geolocation){
      requestBrowserGPS();
      return;
    }
    showGpsPrompt();
  }

        return;
      }
    }catch(_){ }
    hideGpsPrompt();
    try{
      if(navigator.geolocation) requestBrowserGPS();
      else showGpsPrompt();
    }catch(_){showGpsPrompt();}
  }

  function tileUrl(x,y,z){

    if(state.mapLayer==='topo') return 'https://a.tile.opentopomap.org/'+z+'/'+tx+'/'+y+'.png';

  function zoomAt(delta,cx,cy){

    var nx=frac*n,ny=v*n;

    /* Preserve the same map point under the fingers/cursor. */
    var cxWorld=target.x-(cx-w/2), cyWorld=target.y-(cy-h/2);
    var merc=1-2*(cyWorld/n), lat2=180/Math.PI*Math.atan(Math.sinh(Math.PI*merc));

    if(e.target && (e.target.closest && (e.target.closest('#m17UndoPointBtn')||e.target.closest('#m17MyLocationBtn')||e.target.closest('#m17DrawToolbar'))))return;
    mapEl.setPointerCapture&&mapEl.setPointerCapture(e.pointerId);
    if(!state.p1){state.p1={id:e.pointerId,x:e.clientX,y:e.clientY};return;}
    if(!state.p2){state.p2={id:e.pointerId,x:e.clientX,y:e.clientY};return;}
  }
  function pointerMove(e){
    if(!state.p1)return;
    if(state.p2 && e.pointerId===state.p2.id){
      state.p2.x=e.clientX;state.p2.y=e.clientY;
      var a=state.p1,b=state.p2,dx=b.x-a.x,dy=b.y-a.y,d=Math.max(2,Math.hypot(dx,dy));
      state._pinch=state._pinch||{dist:d,zoom:state.zoom,cx:(a.x+b.x)/2,cy:(a.y+b.y)/2};
      var delta=Math.log(d/state._pinch.dist)/Math.LN2;
      var newZ=Math.max(3,Math.min(MAXZ,Math.round(state._pinch.zoom+delta)));
      if(newZ!==state.zoom)zoomAt(newZ-state.zoom,(a.x+b.x)/2-mapEl.getBoundingClientRect().left,(a.y+b.y)/2-mapEl.getBoundingClientRect().top);
      renderSoon();return;
    }
    if(e.pointerId!==state.p1.id)return;
    var dx1=e.clientX-state.p1.x,dy1=e.clientY-state.p1.y;
    if(Math.abs(dx1)+Math.abs(dy1)>1){state.drag=true;applyPan(dx1,dy1);state.p1.x=e.clientX;state.p1.y=e.clientY;renderSoon();}
  }
  function pointerUp(e){
    if(state.p2 && e.pointerId===state.p2.id){state.p2=null;state._pinch=null;return;}
    if(state.p1 && e.pointerId===state.p1.id){
      var wasDrag=state.drag;state.p1=null;state.drag=false;state._pinch=null;
      if(!wasDrag){
        var r=mapEl.getBoundingClientRect(),ll=latLngFromPoint(e.clientX-r.left,e.clientY-r.top);
        if(state.drawMode){
          if(state.drawMode==='line'){
            /* Profile A/B selection must never be mistaken for a double-tap.
               The second tap is the B point and must reach drawTap(). */
            if(window.m4ProfileSelecting===true){
              showCoordAtLatLng(ll.lat,ll.lng);
              drawTap(ll.lat,ll.lng);
            }else if(state.drawPoints.length===0){
              showCoordAtLatLng(ll.lat,ll.lng);
              drawTap(ll.lat,ll.lng);
            }else if(Date.now()-state.lastTap<450){
              finishCurrentDrawing();
              showCoordAtLatLng(ll.lat,ll.lng);
            }else{
              showCoordAtLatLng(ll.lat,ll.lng);
              drawTap(ll.lat,ll.lng);
            }
          }else drawTap(ll.lat,ll.lng);
        }else {
          showCoord(e);
          /* Bridge: send the exact tapped map coordinate directly to project-address flow. */
          if(typeof window.miladgeoProjectPointSelected==='function'){
            try{window.miladgeoProjectPointSelected(ll.lat,ll.lng);}catch(_){}
          }
        }
        var now=Date.now(); if(!state.drawMode && now-state.lastTap<320){zoomAt(1,e.clientX-r.left,e.clientY-r.top);renderSoon();} state.lastTap=now;
      } else renderSoon();
    }
  }

  function m17ParseKmlText(text){
    var parser=new DOMParser(),xml=parser.parseFromString(text,'application/xml');
    if(xml.getElementsByTagName('parsererror').length)throw new Error('فرمت KML معتبر نیست');
    var out=[];
    function directText(el,tag){var n=el.getElementsByTagName(tag)[0];return n?n.textContent.trim():'';}
    function coordsFrom(el){
      var c=directText(el,'coordinates');
      if(!c)return [];
      return c.trim().split(/\s+/).map(function(part){var a=part.split(',');if(a.length<2)return null;var lng=Number(a[0]),lat=Number(a[1]);return isFinite(lat)&&isFinite(lng)?{lat:lat,lng:lng}:null;}).filter(Boolean);
    }
    var placemarks=xml.getElementsByTagName('Placemark'),i,j,pm,lines,polys,pts,name,cs;
    for(i=0;i<placemarks.length;i++){
      pm=placemarks[i]; name=directText(pm,'name');
      pts=pm.getElementsByTagName('Point');
      for(j=0;j<pts.length;j++){cs=coordsFrom(pts[j]);if(cs.length)out.push({type:'point',points:[cs[0]],name:name||''});}
      lines=pm.getElementsByTagName('LineString');
      for(j=0;j<lines.length;j++){cs=coordsFrom(lines[j]);if(cs.length>=2)out.push({type:'line',points:cs,name:name||''});}
      polys=pm.getElementsByTagName('Polygon');
      for(j=0;j<polys.length;j++){
        var outer=polys[j].getElementsByTagName('outerBoundaryIs')[0]||polys[j];
        var rings=outer.getElementsByTagName('LinearRing');
        cs=rings.length?coordsFrom(rings[0]):coordsFrom(polys[j]);
        if(cs.length>=3){
          if(cs.length>2&&Math.abs(cs[0].lat-cs[cs.length-1].lat)<1e-12&&Math.abs(cs[0].lng-cs[cs.length-1].lng)<1e-12)cs.pop();
          out.push({type:'line',points:cs,name:name||''});
        }
      }
    }
    return out;
  }

  function m17ParseGeoJSONText(text){
    var data=JSON.parse(text),out=[];
    function pos(c){if(!Array.isArray(c)||c.length<2)return null;var lng=Number(c[0]),lat=Number(c[1]);return isFinite(lat)&&isFinite(lng)&&Math.abs(lat)<=90&&Math.abs(lng)<=180?{lat:lat,lng:lng}:null;}
    function addCoords(type,coords,name){
      if(type==='Point'){var p=pos(coords);if(p)out.push({type:'point',points:[p],name:name||''});return;}
      if(type==='MultiPoint'){(coords||[]).forEach(function(c){var p=pos(c);if(p)out.push({type:'point',points:[p],name:name||''});});return;}
      if(type==='LineString'){var pts=(coords||[]).map(pos).filter(Boolean);if(pts.length>=2)out.push({type:'line',points:pts,name:name||''});return;}
      if(type==='MultiLineString'){(coords||[]).forEach(function(c){addCoords('LineString',c,name);});return;}
      if(type==='Polygon'){var ring=coords&&coords[0]||[],pts=ring.map(pos).filter(Boolean);if(pts.length>2){if(pts.length>2&&pts[0].lat===pts[pts.length-1].lat&&pts[0].lng===pts[pts.length-1].lng)pts.pop();if(pts.length>=3)out.push({type:'line',measureType:'area',points:pts,name:name||''});}return;}
      if(type==='MultiPolygon'){(coords||[]).forEach(function(poly){addCoords('Polygon',poly,name);});return;}
      if(type==='GeometryCollection'){(coords||[]).forEach(function(g){if(g)addCoords(g.type,g.coordinates||g.geometries,name);});}
    }
    function feature(f){if(!f)return;var name=(f.properties&&(f.properties.name||f.properties.Name||f.properties.title))||'';if(f.type==='FeatureCollection')(f.features||[]).forEach(feature);else if(f.type==='Feature')feature({type:'Geometry',geometry:f.geometry,properties:f.properties});else if(f.type==='Geometry') {if(f.geometry)addCoords(f.geometry.type,f.geometry.coordinates||f.geometry.geometries,name);}else if(f.type==='GeometryCollection')addCoords('GeometryCollection',f.geometries,name);else if(f.type&&f.coordinates!==undefined)addCoords(f.type,f.coordinates,name);}
    feature(data);return out;
  }

  function m17ParseShapefile(buffer){
    var dv=new DataView(buffer),out=[];if(buffer.byteLength<100||dv.getInt32(0,false)!==9994)throw new Error('فایل SHP معتبر نیست');
    function xy(off){var x=dv.getFloat64(off,true),y=dv.getFloat64(off+8,true);if(!isFinite(x)||!isFinite(y)||Math.abs(y)>90||Math.abs(x)>180) return null;return {lat:y,lng:x};}
    var off=100;while(off+8<=buffer.byteLength){var words=dv.getInt32(off+4,false),len=words*2,base=off+8;if(len<4||base+len>buffer.byteLength)break;var typ=dv.getInt32(base,true);
      if(typ===1||typ===11||typ===21){var p=xy(base+4);if(p)out.push({type:'point',points:[p]});}
      else if(typ===8||typ===18||typ===28){var n=dv.getInt32(base+36,true),q=base+40,pts=[];if(n>=0&&n<1000000&&q+n*16<=base+len)for(var a=0;a<n;a++){var pp=xy(q+a*16);if(pp)pts.push(pp);}if(pts.length)out.push({type:'line',points:pts});}
      else if(typ===3||typ===5||typ===13||typ===15||typ===23||typ===25){var np=dv.getInt32(base+36,true),npt=dv.getInt32(base+40,true),partOff=base+44,ptOff=partOff+np*4;if(np>=0&&np<1000000&&npt>=0&&npt<5000000&&ptOff+npt*16<=base+len){var starts=[];for(var z=0;z<np;z++)starts.push(dv.getInt32(partOff+z*4,true));starts.push(npt);for(var r=0;r<starts.length-1;r++){var arr=[];for(var h=starts[r];h<starts[r+1];h++){var xypt=xy(ptOff+h*16);if(xypt)arr.push(xypt);}if(typ===5||typ===15||typ===25){if(arr.length>2&&arr[0].lat===arr[arr.length-1].lat&&arr[0].lng===arr[arr.length-1].lng)arr.pop();if(arr.length>=3)out.push({type:'line',measureType:'area',points:arr});}else if(arr.length>=2)out.push({type:'line',points:arr});}}}
      off=base+len;
    }return out;
  }

      else if(g.type==='LineString'&&c.length>=2){xml.push('<trk><name>'+esc(f.properties.name)+'</name><trkseg>');for(j=0;j<c.length;j++)xml.push('<trkpt lat="'+c[j][1]+'" lon="'+c[j][0]+'"></trkpt>');xml.push('</trkseg></trk>');}
      else if(g.type==='Polygon'&&c[0]&&c[0].length>=4){xml.push('<trk><name>'+esc(f.properties.name)+'</name><trkseg>');for(j=0;j<c[0].length;j++)xml.push('<trkpt lat="'+c[0][j][1]+'" lon="'+c[0][j][0]+'"></trkpt>');xml.push('</trkseg></trk>');}
    }
    xml.push('</gpx>');
    if(!features.length){showToast('برای خروجی، ابتدا نقطه یا ترسیم اضافه کنید');return;}
    var ok=m17DownloadBlob(new Blob([xml.join('\n')],{type:'application/gpx+xml;charset=utf-8'}),'miladgeo-menu17.gpx');
    if(!ok)showToast('دانلود GPX انجام نشد');
  }

  function m17LoadJsZip(cb){
    if(window.JSZip&&typeof window.JSZip.loadAsync==='function'){cb(window.JSZip);return;}
    window.__m17ZipWaiters=window.__m17ZipWaiters||[];
    window.__m17ZipWaiters.push(cb);
    if(window.__m17ZipLoading)return;
    window.__m17ZipLoading=true;
    var urls=['./jszip.min.js','file:///android_asset/jszip.min.js','https://cdnjs.cloudflare.com/ajax/libs/jszip/3.10.1/jszip.min.js'],i=0;
    function finish(lib){window.__m17ZipLoading=false;var q=window.__m17ZipWaiters||[];window.__m17ZipWaiters=[];q.forEach(function(fn){try{fn(lib||null);}catch(e){console.error('M17 JSZip callback',e);}});}
    function next(){
      if(window.JSZip&&typeof window.JSZip.loadAsync==='function'){finish(window.JSZip);return;}
      if(i>=urls.length){finish(null);return;}
      var src=urls[i++],sc=document.createElement('script');sc.src=src;sc.async=true;
      sc.onload=function(){if(window.JSZip&&typeof window.JSZip.loadAsync==='function')finish(window.JSZip);else next();};
      sc.onerror=next;document.head.appendChild(sc);
    }
    next();
  }
