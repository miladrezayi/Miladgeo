/* V482: infer PENZD by whole-column patterns, never by digit-count alone for P or Z. */
var excelDetectedMapping=null;
function excelHeaderRole(h){
 var n=excelNorm(h).replace(/[آأإ]/g,'ا');
 if(/^(p|pt|point|pointno|pointnumber|pointid|station|stationid|counter|count|no|number|id|شماره|شمارنده|شماره‌نقطه|شمارهنقطه|نقطه|شناسه)$/.test(n))return 'P';
 if(/^(x|e|easting|east|xcoord|xcoordinate|مختصاتx|مختصاتشرقی|شرقی)$/.test(n))return 'E';
 if(/^(y|n|northing|north|ycoord|ycoordinate|مختصاتy|مختصاتشمالی|شمالی)$/.test(n))return 'N';
 if(/^(z|elevation|elev|height|level|rl|ارتفاع|تراز)$/.test(n))return 'Z';
 if(/^(dp|description|desc|code|pointcode|type|کد|شرح|توضیحات|توصیحات)$/.test(n))return 'D';
 return '';
}
function excelColumnStats(col,rows){
 var vals=[],nums=[],ints=[],uniq={},seq=0,inc=0,digits=[],diffs=[],i,v,n,prev;
 for(i=0;i<rows.length;i++){v=rows[i]&&rows[i][col]!=null?rows[i][col]:'';if(String(v).trim()==='')continue;vals.push(v);n=excelNum(v);if(isFinite(n)){nums.push(n);if(Math.abs(n-Math.round(n))<1e-9)ints.push(n);uniq[String(v).trim()]=1;}}
 for(i=0;i<ints.length;i++){var a=Math.abs(ints[i]);digits.push(a===0?1:String(Math.floor(a)).length);if(i){var d=ints[i]-ints[i-1];diffs.push(d);if(d>0)inc++;if(d===1)seq++;}}
 var sorted=nums.slice().sort(function(a,b){return a-b;}),median=sorted.length?sorted[Math.floor(sorted.length/2)]:NaN;
 var avgDigits=digits.length?digits.reduce(function(a,b){return a+b;},0)/digits.length:0,stepConsistency=0,medianStep=NaN;if(diffs.length){var absDiffs=diffs.map(function(x){return Math.abs(x);}).sort(function(a,b){return a-b;});medianStep=absDiffs[Math.floor(absDiffs.length/2)];var stepCounts={};for(i=0;i<diffs.length;i++){var sk=String(Math.round(diffs[i]*1000)/1000);stepCounts[sk]=(stepCounts[sk]||0)+1;}stepConsistency=Math.max.apply(null,Object.keys(stepCounts).map(function(k){return stepCounts[k];}))/diffs.length;}
 return {index:col,values:vals.length,nums:nums.length,numRatio:vals.length?nums.length/vals.length:0,intRatio:nums.length?ints.length/nums.length:0,uniqueRatio:ints.length?Object.keys(uniq).length/ints.length:0,seqRatio:ints.length>1?seq/(ints.length-1):0,incRatio:ints.length>1?inc/(ints.length-1):0,stepConsistency:stepConsistency,medianStep:medianStep,median:median,digits:avgDigits,unique:Object.keys(uniq).length};
}
function excelDetectColumns(){
 var rows=excelRows||[],headers=excelHeaders||[],count=headers.length;
 for(var ri=0;ri<rows.length;ri++)count=Math.max(count,(rows[ri]||[]).length);
 var stats=[],used={},map={P:-1,E:-1,N:-1,Z:-1,D:-1},source={},i,role;
 for(i=0;i<count;i++){
  stats[i]=excelColumnStats(i,rows);
  role=($('excelHeaderMode')&&$('excelHeaderMode').value==='YES')?excelHeaderRole(headers[i]||''):'';
  if(role&&map[role]===-1){map[role]=i;used[i]=true;source[role]='عنوان ستون';}
 }
 function best(candidates,scoreFn){var bi=-1,bs=-1e9;for(var k=0;k<candidates.length;k++){var ix=candidates[k],sc=scoreFn(stats[ix]);if(sc>bs){bs=sc;bi=ix;}}return bi;}
 var available=[];for(i=0;i<count;i++)if(!used[i]&&stats[i].values)available.push(i);
 /* P is identified by counter/sequence behaviour and uniqueness, not digit length. */
 if(map.P<0){var p=best(available,function(a){return 4*a.intRatio+3*a.uniqueRatio+4*a.seqRatio+1.5*a.incRatio-0.15*Math.max(0,a.digits-7);});if(p>=0&&stats[p].numRatio>=0.75&&stats[p].intRatio>=0.75&&stats[p].uniqueRatio>=0.55&&(stats[p].seqRatio>=0.5||(stats[p].stepConsistency>=0.85&&stats[p].medianStep<=100&&stats[p].digits<=5))){map.P=p;used[p]=true;source.P='الگوی شمارنده';}}
 available=[];for(i=0;i<count;i++)if(!used[i]&&stats[i].values)available.push(i);
 /* Coordinate columns: six-digit Easting and seven-digit Northing are clues, not hard rules. */
 if(map.E<0&&available.length){var e=best(available,function(a){var digitFit=Math.max(0,1-Math.abs(a.digits-6)/4);var mag=isFinite(a.median)?Math.max(0,1-Math.abs(Math.log10(Math.max(1,Math.abs(a.median)))-5.7)/2):0;return 3*a.numRatio+2*a.intRatio+2*digitFit+mag;});if(e>=0&&stats[e].numRatio>=0.65){map.E=e;used[e]=true;source.E='الگوی مختصات شرقی';}}
 available=[];for(i=0;i<count;i++)if(!used[i]&&stats[i].values)available.push(i);
 if(map.N<0&&available.length){var n=best(available,function(a){var digitFit=Math.max(0,1-Math.abs(a.digits-7)/4);var mag=isFinite(a.median)?Math.max(0,1-Math.abs(Math.log10(Math.max(1,Math.abs(a.median)))-6.5)/2):0;return 3*a.numRatio+2*a.intRatio+2*digitFit+mag;});if(n>=0&&stats[n].numRatio>=0.65){map.N=n;used[n]=true;source.N='الگوی مختصات شمالی';}}
 available=[];for(i=0;i<count;i++)if(!used[i]&&stats[i].values)available.push(i);
 /* Z is the remaining numeric measurement; its digit count is deliberately ignored. */
 if(map.Z<0&&available.length){var z=best(available,function(a){var magnitude=isFinite(a.median)?Math.abs(Math.log10(Math.max(1,Math.abs(a.median)))):99;return 4*a.numRatio+2*a.intRatio-0.55*magnitude;});if(z>=0&&stats[z].numRatio>=0.45){map.Z=z;used[z]=true;source.Z='ستون عددی باقی‌مانده / ارتفاع';}}
 available=[];for(i=0;i<count;i++)if(!used[i]&&stats[i].values)available.push(i);
 if(map.D<0&&available.length){var d=best(available,function(a){return 4*(1-a.numRatio)+a.values/Math.max(1,rows.length);});map.D=d;used[d]=true;source.D='متن / کد باقی‌مانده';}
 /* Positional format is a last-resort fallback only for roles still unresolved. */
 var fmt=$('excelFormat')?$('excelFormat').value:'PENZD',pos={PENZD:[0,1,2,3,4],PNEZD:[0,2,1,3,4],NEZD:[-1,1,0,2,3],ENZD:[-1,0,1,2,3]}[fmt]||[0,1,2,3,4];
 var roles=['P','E','N','Z','D'];
 for(i=0;i<roles.length;i++){role=roles[i];if(map[role]<0&&pos[i]>=0&&pos[i]<count&&!used[pos[i]]){map[role]=pos[i];used[pos[i]]=true;source[role]='ترتیب قالب انتخابی (پیشنهادی)';}}
 map.stats=stats;map.source=source;map.confidence=['P','E','N','Z'].filter(function(k){return map[k]>=0&&source[k]&&source[k]!=='ترتیب قالب انتخابی (پیشنهادی)';}).length;
 excelDetectedMapping=map;return map;
}

function importExcelPoints(silent){
 silent=!!silent;

 if(!a.length){if(!silent)showToast('No valid X / Y rows found');return false;}
 /* V479: Excel import is additive. Preview never changes MAIN POINT LIST.
    A row is duplicate only when its point number OR its exact X/Y/Z already exists.
    New rows are always appended and immediately saved/refreshed. */
 var existingNo={},existingCoord={},seenNo={},seenCoord={},clean=[];
 for(var ei=0;ei<points.length;ei++){
   var ep=points[ei],eno=String(ep.no||'').trim().toUpperCase();
   var ecoord=[Number(ep.x).toFixed(6),Number(ep.y).toFixed(6),Number(ep.z).toFixed(6)].join('|');
   if(eno)existingNo[eno]=true;
   existingCoord[ecoord]=true;
 }
 for(var di=0;di<a.length;di++){
   var ap0=a[di],no0=String(ap0.no||'').trim().toUpperCase();
   var coord0=[Number(ap0.x).toFixed(6),Number(ap0.y).toFixed(6),Number(ap0.z).toFixed(6)].join('|');
   if((no0&&seenNo[no0])||seenCoord[coord0])continue;
   seenNo[no0]=true;seenCoord[coord0]=true;clean.push(ap0);
 }
 a=clean;
 var added=0,dupes=0,conflicts=0;
 for(var i=0;i<a.length;i++){
   var ap=a[i],noKey=String(ap.no||'').trim().toUpperCase();
   var coordKey=[Number(ap.x).toFixed(6),Number(ap.y).toFixed(6),Number(ap.z).toFixed(6)].join('|');
   if(existingCoord[coordKey]){dupes++;continue;}
   if(noKey&&existingNo[noKey]){conflicts++;continue;}
   points.push(ap);
   excelImportedIds.push(ap.id);
   if(noKey)existingNo[noKey]=true;
   existingCoord[coordKey]=true;
   added++;

function parseDXF(text){
 var p=dxfPairs(text),ents=[],ent=null,poly=null;
 for(var i=0;i<p.length;i++){
  var c=p[i].code,v=p[i].value;
  if(c==='0'){
   if(ent){
    if(ent.type!=='VERTEX')ents.push(ent);
    ent=null;
   }
   if(v==='POINT')ent={type:'POINT',pts:[],layer:'',z:0};
   else if(v==='LINE')ent={type:'LINE',pts:[],layer:'',z:0};
   else if(v==='LWPOLYLINE')ent={type:'LWPOLYLINE',pts:[],closed:false,layer:'',z:0};
   else if(v==='POLYLINE')ent={type:'POLYLINE',pts:[],closed:false,layer:'',z:0};
   else if(v==='SEQEND'){if(poly){ents.push(poly);poly=null;}}
   else if(v==='VERTEX')ent={type:'VERTEX',pts:[],layer:'',z:0};
  }else if(ent){
   if(c==='8')ent.layer=String(v);
   else if(c==='10')ent._x=parseFloat(v);
   else if(c==='20')ent._y=parseFloat(v);
   else if(c==='30')ent._z=parseFloat(v);
   else if(c==='38')ent._z=parseFloat(v);
   else if(c==='11')ent._x2=parseFloat(v);
   else if(c==='21')ent._y2=parseFloat(v);
   else if(c==='31')ent._z2=parseFloat(v);
   else if(c==='70')ent.closed=((parseInt(v)||0)&1)?true:false;
   if((ent.type==='LWPOLYLINE'||ent.type==='VERTEX')&&c==='10'){
    ent.pts.push({x:parseFloat(v),y:NaN,z:0});
   }
   if((ent.type==='LWPOLYLINE'||ent.type==='VERTEX')&&c==='20'&&ent.pts.length){
    ent.pts[ent.pts.length-1].y=parseFloat(v);
   }
   if((ent.type==='LWPOLYLINE'||ent.type==='VERTEX')&&c==='30'&&ent.pts.length){
    ent.pts[ent.pts.length-1].z=parseFloat(v);
   }
   if(ent.type==='POINT'&&c==='20'){
    ent.pts=[{x:ent._x,y:parseFloat(v),z:Number.isFinite(ent._z)?ent._z:0}];
   }
   if(ent.type==='LINE'&&c==='21'){
    ent.pts=[{x:ent._x,y:ent._y,z:Number.isFinite(ent._z)?ent._z:0},{x:ent._x2,y:parseFloat(v),z:Number.isFinite(ent._z2)?ent._z2:0}];
   }
  }
 }

 return ents.filter(function(e){return e.pts&&e.pts.length;});
}
